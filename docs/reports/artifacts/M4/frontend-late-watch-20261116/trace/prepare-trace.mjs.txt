import {readFileSync,writeFileSync} from 'node:fs';
import path from 'node:path';
import {fileURLToPath} from 'node:url';
const here=path.dirname(fileURLToPath(import.meta.url));
const prior=path.join(here,'archive/prior-diagnostic');
const change=(text,from,to)=>{if(text.split(from).length!==2)throw new Error('Trace patch must have exactly one match: '+from);return text.replace(from,to);};
let frontend=readFileSync(path.join(prior,'entry-frontends.mjs'),'utf8');
frontend=change(frontend,"import {appendFileSync} from 'node:fs';","import {installWatcherTrace} from './watcher-observer.mjs';\nimport {appendFileSync} from 'node:fs';");
frontend=change(frontend,"const diagFile=process.env.FRONTEND_DIAG_LOG;if(!diagFile)throw new Error('Missing diagnostic log');","const diagFile=process.env.FRONTEND_DIAG_LOG;if(!diagFile)throw new Error('Missing diagnostic log');\nconst watcherTrace=installWatcherTrace({file:diagFile+'.watchers.jsonl'});");
frontend=change(frontend,"map(handle=>({type:handle.constructor?.name,","map(handle=>({watcher:watcherTrace.identify(handle),type:handle.constructor?.name,");
frontend=change(frontend,"function resources(){return {handles:handles(),","function resources(){return {watchers:watcherTrace.snapshot(),handles:handles(),");
frontend=change(frontend,String.raw`const log=(event,value={})=>appendFileSync(diagFile,JSON.stringify({at:new Date().toISOString(),pid:process.pid,event,...value})+'\n');`,String.raw`const log=(event,value={})=>{watcherTrace.setPhase(event,value.label);appendFileSync(diagFile,JSON.stringify({at:new Date().toISOString(),pid:process.pid,event,...value})+'\n');};`);
writeFileSync(path.join(here,'frontends-diagnostic.mjs'),frontend);
writeFileSync(path.join(here,'launch-diagnostic.mjs'),readFileSync(path.join(prior,'entry-launch.mjs')));
let driver=readFileSync(path.join(prior,'diagnose-once.mjs.txt'),'utf8');
driver=change(driver,"limits:{cases:3,startupMs:75000,closeMs:105000,unchangedLaunchCloseMs:45000}","limits:{cases:1,startupMs:75000,closeMs:105000,unchangedLaunchCloseMs:45000},traceScope:'One consumed case only; delegated watcher creation/lifecycle observation; no changes to watcher lifetime'");
driver=change(driver,"for(const name of ['prepare-copies.mjs','diagnose-once.mjs'])writeFileSync", "for(const name of ['prepare-trace.mjs','watcher-observer.mjs','diagnose-once.mjs'])writeFileSync");
driver=change(driver,"['frontends-diagnostic.mjs','launch-diagnostic.mjs'].map", "['frontends-diagnostic.mjs','launch-diagnostic.mjs','watcher-observer.mjs'].map");
driver=change(driver,"No-request control performs none of these requests.","Only the consumed scenario executes; no request/status-only variants are rerun. Delegated fs.watch/fs.watchFile/unwatchFile observation is installed before Vite import; watcher ref state read-only.");
driver=change(driver,"for(const kind of ['no-request','consumed','status-only']){", "for(const kind of ['consumed']){");
writeFileSync(path.join(here,'diagnose-once.mjs'),driver);
console.log('Prepared one consumed watcher trace; no services started');
