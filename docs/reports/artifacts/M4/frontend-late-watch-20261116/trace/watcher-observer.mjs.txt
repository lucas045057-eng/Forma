import fs from 'node:fs';
import path from 'node:path';
import {fileURLToPath} from 'node:url';

export function installWatcherTrace({file}){

  write('watch-created',{id,api,path:record.path,options:record.options,callsite,state:nativeState(object),weakObjectRegistry:true});

   const original=object[key];if(typeof original!=='function')continue;const descriptor=Object.getOwnPropertyDescriptor(object,key);
   try{Object.defineProperty(object,key,{configurable:true,writable:true,enumerable:descriptor?.enumerable??false,value:function(...parameters){if(key==='close')record.closeCalls++;if(key==='ref')record.refCalls++;if(key==='unref')record.unrefCalls++;write('watch-method-call',{id,method:key,state:nativeState(this),callsite:stack()});try{const result=Reflect.apply(original,this,parameters);write('watch-method-return',{id,method:key,returnsSameObject:result===this,state:nativeState(this)});return result;}catch(error){write('watch-method-throw',{id,method:key,error:{name:error.name,code:error.code,message:error.message}});throw error;}}});}catch(error){write('observer-wrap-error',{id,method:key,message:error.message});}
  }
  if(typeof object.on==='function')for(const event of ['close','stop'])object.on(event,function(){if(event==='close')record.closeEventSeen=true;else record.stopEventSeen=true;write('watch-event',{id,type:event,state:nativeState(this),callsite:stack()});});
  const emit=object.emit;if(typeof emit==='function')try{Object.defineProperty(object,'emit',{configurable:true,writable:true,enumerable:false,value:function(event,...parameters){if(['close','stop','error'].includes(event))write('watch-emit',{id,type:event,error:event==='error'?{name:parameters[0]?.name,code:parameters[0]?.code,message:parameters[0]?.message}:undefined,state:nativeState(this),callsite:stack()});return Reflect.apply(emit,this,[event,...parameters]);}});}catch(error){write('observer-wrap-error',{id,method:'emit',message:error.message});}
  return id;
 }
 for(const key of ['watch','watchFile']){const original=fs[key];fs[key]=function(...args){const callsite=stack();write('watch-call',{api:key,path:safePath(args[0]),callsite});let object;try{object=Reflect.apply(original,this,args);}catch(error){write('watch-throw',{api:key,path:safePath(args[0]),error:{name:error.name,code:error.code,message:error.message},callsite});throw error;}try{track(object,key,args,callsite);}catch(error){write('observer-track-error',{api:key,message:error.message,callsite});}return object;};}

 syncBuiltinESMExports();write('observer-installed',{syncBuiltinESMExports:true,scope:'Diagnostic process only; delegated fs calls preserve this/args/result/errors; no forced watcher close/ref changes',initial:snapshot()});
 return{identify,snapshot,setPhase(event,label){phase={event,label};if(['owned-close-begin','owned-close-finished','process:disconnect','process:beforeExit','process:exit'].includes(event))write('watcher-snapshot',{reason:event,...snapshot()});}};
}
