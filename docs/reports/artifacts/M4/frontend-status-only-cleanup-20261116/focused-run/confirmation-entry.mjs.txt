import {spawn,execFileSync} from 'node:child_process';

import {createHash} from 'node:crypto';
import path from 'node:path';
import {fileURLToPath} from 'node:url';
import {systemEnvironment} from '../scripts/demo/safety.mjs';
const root=path.resolve(path.dirname(fileURLToPath(import.meta.url)),'..');const env=systemEnvironment();
const git=(...args)=>execFileSync('git',['-c','core.longpaths=true','-C',root,...args],{encoding:'utf8',env});
const out=path.join(root,'.data','frontend-close-focused-'+new Date().toISOString().replaceAll(':','-'));mkdirSync(out);

const identity=()=>Object.fromEntries(names.map(name=>[name,sha(readFileSync(path.join(root,name)))]));

child.once('close',async(code,signal)=>{
 record.code=code;record.signal=signal;record.finished=new Date().toISOString();
 await Promise.all([new Promise(resolve=>stdout.closed?resolve():stdout.once('close',resolve)),new Promise(resolve=>stderr.closed?resolve():stderr.once('close',resolve))]);
 try{record.identityAfter=identity();record.changed=names.filter(name=>record.identityBefore[name]!==record.identityAfter[name]);record.headAfter=git('rev-parse','HEAD').trim();record.statusAfter=git('status','--porcelain');const tap=readFileSync(path.join(out,'stdout.log'),'utf8');record.actualCounts={};for(const label of ['tests','pass','fail','cancelled','skipped','todo']){const matches=[...tap.matchAll(new RegExp('^# '+label+' (\\d+)\\s*$','gm'))];record.actualCounts[label]=matches.length?Number(matches.at(-1)[1]):null;}record.caseResults={};record.notExecuted=[];for(const kind of ['no-request','consumed','status-only']){try{const result=JSON.parse(readFileSync(path.join(out,kind,'result.json'),'utf8'));record.caseResults[kind]={closed:result.closed,status:result.after?.status,primary:result.primary??null,cleanupErrors:result.cleanupErrors,remaining:result.remaining,fallbackKills:result.fallbackKills,hmr:result.hmr};}catch(error){record.notExecuted.push({kind,reason:error.message});}}
 }catch(error){record.errors.push({stage:'finalize',message:error.message,stack:error.stack});}
 const counts=record.actualCounts??{};record.accepted=code===0&&signal===null&&!record.errors.length&&!record.changed?.length&&record.headAfter===record.head&&counts.tests===record.expectedFromSource.contracts+record.expectedFromSource.realScenarios&&counts.pass===counts.tests&&counts.fail===0&&counts.cancelled===0&&counts.skipped===0&&counts.todo===0&&record.notExecuted?.length===0;save();
 const payloads=[];function collect(dir,prefix=''){for(const entry of readdirSync(dir,{withFileTypes:true})){const rel=path.join(prefix,entry.name);if(entry.isDirectory())collect(path.join(dir,entry.name),rel);else if(entry.isFile())payloads.push({file:rel.replaceAll('\\','/'),sha256:sha(readFileSync(path.join(dir,entry.name)))});}}collect(out);writeFileSync(path.join(out,'sha256-manifest.json'),JSON.stringify({excludesSelf:true,files:payloads},null,2)+'\n');
 console.log(JSON.stringify({run:out,head:record.head,code,signal,actualCounts:record.actualCounts,changed:record.changed,notExecuted:record.notExecuted,accepted:record.accepted,errors:record.errors}));process.exitCode=record.accepted?0:1;
});
console.log(JSON.stringify({run:out,head:record.head,pid:child.pid,command:record.command,scope:'Focused gate contracts plus three true production HTTP shutdown cases; live HMR included in consumed; no original diagnostic copies'}));
